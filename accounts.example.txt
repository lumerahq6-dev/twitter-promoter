# so trailing fields may be omitted.
#
#   username:password:email:email_password:auth_token:totp_secret
#
#   auth_token   40-char hex cookie from a logged-in browser session (optional if
#                a password login is possible)
#   totp_secret  base32 2FA seed (optional)
#
# Copy to accounts.txt, which is gitignored. Never commit real credentials.
